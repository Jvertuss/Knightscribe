cmake_minimum_required(VERSION 3.16)
project(knightcribe CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)   # optimized build (matters a lot on the Pi)
endif()
find_package(OpenCV REQUIRED)
find_package(Threads REQUIRED)
add_executable(knightcribe knightcribe.cpp)
target_link_libraries(knightcribe ${OpenCV_LIBS} Threads::Threads)
